# Root-relative paths left out of compiler/bundle.zip (fnmatch globs; `*` also matches `/`).
# Everything here is unused by Tectonic (XeTeX engine, LaTeX release kernel) and would otherwise
# collide in the flat bundle with a file of the same basename.

# Documentation, sources and scripts.
doc/*
source/*
scripts/*
*/README
*/README.*
*/README-*

# LaTeX pre-release kernel; the bundle ships the released kernel from tex/latex.
tex/latex-dev/*

# Lua code and LuaTeX-only trees: Tectonic runs XeTeX, not LuaTeX.
*.lua
tex/luatex/*
tex/lualatex/*

# dvipdfm(x) and dvips-only font maps; pdftex.map/updmap output is generated separately.
fonts/map/dvipdfm/*
fonts/cmap/dvipdfmx/*

# Metafont sources and the mmap package's duplicate CMaps (cmap ships the same encodings).
mft/*
tex/latex/mmap/*

# Per-faculty locale copies of fithesis (the generic locale files stay).
tex/latex/fithesis/locale/mu/*

# Copies of the PGF manual helpers bundled by tikz-dependency.
tex/latex/tikz-dependency/pgfmanual*

# Size budget (npm rejects very large packages). XeTeX defaults to Latin Modern OpenType, and
# Libertine/TeX Gyre ship OpenType files too, so these Type1 copies are only used by 8-bit T1
# setups the writer avoids. Latin Modern Type1 (fonts/type1/public/lm) stays for `lmodern`.
fonts/type1/public/cm-super/*
fonts/type1/public/libertine/*
fonts/type1/public/tex-gyre/*
# AFM metrics are for dvips/pdfTeX only; Flash files are never used.
*.afm
*.swf
# Packages whose payload is large bitmap/PDF art.
tex/latex/twemojis/*
tex/latex/hwemoji/*
tex/latex/realhats/*
# AFM metrics (also gzipped) are for dvips/pdfTeX only.
fonts/afm/*
# MetaPost data (Tectonic runs no MetaPost) and emoji-as-TikZ art.
metapost/*
tex/latex/utfsym/*
